#Data-Preprocessing-Correlation-Analysis-Pipeline

import pandas as pd
from sklearn.preprocessing import MinMaxScaler, StandardScaler, RobustScaler
import numpy as np
from scipy.stats import pearsonr

# Read the Excel file
input_file = "Dynamic_Table_m.xlsx"  # Replace with your file name
df = pd.read_excel(input_file)

# Handle missing values (fill numeric columns with the column mean)
numeric_columns = df.select_dtypes(include=["float64", "int64"]).columns
df[numeric_columns] = df[numeric_columns].fillna(df[numeric_columns].mean())

# Calculate descriptive statistics before outlier removal
descriptive_stats_before = df[numeric_columns].describe()
print("Descriptive statistics (before outlier removal):")
print(descriptive_stats_before)

# Identify and remove outliers using the IQR method (numeric columns only)
for column in numeric_columns:
    Q1 = df[column].quantile(0.25)
    Q3 = df[column].quantile(0.75)
    IQR = Q3 - Q1
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR
    df = df[(df[column] >= lower_bound) & (df[column] <= upper_bound)]

# Calculate descriptive statistics after outlier removal
descriptive_stats_after = df[numeric_columns].describe()
print("Descriptive statistics (after outlier removal):")
print(descriptive_stats_after)

# Normalize the numeric columns (excluding categorical ones like "Year" and "Country")
scaler = StandardScaler()
df[numeric_columns] = scaler.fit_transform(df[numeric_columns])

# Save the cleaned and normalized dataset to a new Excel file
output_file = "processed_datam.xlsx"
df.to_excel(output_file, index=False)
print(f"Processed data saved to {output_file}")

# Define a function to calculate p-values for the correlation matrix
def calculate_p_values(df, numeric_columns):
    df_numeric = df[numeric_columns]
    p_values = pd.DataFrame(index=numeric_columns, columns=numeric_columns)
    for col1 in numeric_columns:
        for col2 in numeric_columns:
            if col1 == col2:  # Perfect correlation (1.0) with itself
                p_values.loc[col1, col2] = 0
            else:
                _, p_value = pearsonr(df_numeric[col1], df_numeric[col2])
                p_values.loc[col1, col2] = p_value
    return p_values

# Calculate correlation matrix (numeric columns only)
numeric_columns = ["FDI", "Mobile_BB", "Fixed_BB", "4G deployment", "inflation rate", "unemployment", "ICT index"]
correlation_matrix = df[numeric_columns].corr()
print("Correlation Matrix:")
print(correlation_matrix)

# Calculate p-values for statistical significance
p_values = calculate_p_values(df, numeric_columns)
print("\nP-Values for Statistical Significance:")
print(p_values)

# Combine the correlation matrix and p-values, adding stars for significance
correlation_summary = pd.DataFrame(index=numeric_columns, columns=numeric_columns)
for row in numeric_columns:
    for col in numeric_columns:
        correlation = correlation_matrix.loc[row, col]
        p_value = float(p_values.loc[row, col])
        significance = ""
        if p_value < 0.01:  # Add stars for statistical significance at 1% level
            significance = "***"
        correlation_summary.loc[row, col] = f"{correlation:.2f} (p={p_value:.3f}) {significance}"

print("\nCorrelation Matrix with P-Values:")
print(correlation_summary)
